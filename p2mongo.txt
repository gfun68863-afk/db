Queries
1. Display total count of movies released in year 2020
db.movies.countDocuments({
    release_date: {
        $gte: ISODate("2020-01-01"),
        $lt: ISODate("2021-01-01")
    }
})

2. Display movie names having both International and National awards
db.movies.find(
{
    awards: {
        $all: ["International", "National"]
    }
},
{
    movie_name: 1,
    _id: 0
}
)

3. Display all movies with rating more than 4.2
db.movies.find({
    rating_star: { $gt: 4.2 }
})

4. Display movie names whose collection is less than budget
db.movies.find(
{
    $expr: {
        $lt: ["$collection_amount", "$budget"]
    }
},
{
    movie_name: 1,
    _id: 0
}
)

5. Delete all movies with rating < 2.0
db.movies.deleteMany({
    rating_star: { $lt: 2.0 }
})

6. Provide International award to movies with rating > 4.8
db.movies.updateMany(
{
    rating_star: { $gt: 4.8 }
},
{
    $addToSet: {
        awards: "International"
    }
}
)
