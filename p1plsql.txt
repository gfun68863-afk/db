CREATE TABLE Student (
    Rollno NUMBER PRIMARY KEY,
    Name VARCHAR2(50),
    DateOfAdmission DATE,
    Branch VARCHAR2(30),
    Percent NUMBER(5,2),
    Status VARCHAR2(20)
);

CREATE TABLE Alumni (
    Rollno NUMBER,
    Name VARCHAR2(50),
    DateOfAdmission DATE,
    Branch VARCHAR2(30),
    Percent NUMBER(5,2),
    Status VARCHAR2(20),
    Action_Type VARCHAR2(20),
    Action_Date DATE
);


CREATE OR REPLACE TRIGGER trg_student
BEFORE UPDATE OR DELETE
ON Student
FOR EACH ROW

BEGIN

    INSERT INTO Alumni
    VALUES(
        :OLD.Rollno,
        :OLD.Name,
        :OLD.DateOfAdmission,
        :OLD.Branch,
        :OLD.Percent,
        :OLD.Status,
        
        CASE
            WHEN UPDATING THEN 'UPDATED'
            WHEN DELETING THEN 'DELETED'
        END,
        
        SYSDATE
    );

END;

6. Create Departments Table

7. Create Professors Table
CREATE TABLE Professors (
    prof_id NUMBER PRIMARY KEY,
    prof_fname VARCHAR2(30),
    prof_lname VARCHAR2(30),
    dept_id NUMBER,
    designation VARCHAR2(30),
    salary NUMBER,
    doj DATE,
    email VARCHAR2(50),
    phone VARCHAR2(15),
    city VARCHAR2(30)
);

8. Create Works Table
CREATE TABLE Works (

    duration NUMBER
);

9. Create Shift Table
CREATE TABLE Shift (
    prof_id NUMBER,

QUERY 2
Display professors from Pune/Mumbai and name starts with A or D
SELECT *
FROM Professors
WHERE city IN ('Pune','Mumbai')
AND (prof_fname LIKE 'A%' OR prof_fname LIKE 'D%');

QUERY 3
Number of different cities
SELECT COUNT(DISTINCT city) AS Total_Cities
FROM Professors;

QUERY 4
Give 5% salary increment
UPDATE Professors
SET salary = salary + (salary * 0.05)
WHERE doj = TO_DATE('01-01-2015','DD-MM-YYYY');

QUERY 5
Delete professors from Pune
DELETE FROM Professors
WHERE city='Pune';

QUERY 10
Professor + Department Details
SELECT p.*, d.*
FROM Professors p
JOIN Departments d
ON p.dept_id = d.dept_id;

QUERY 11
prof_id, name and shift
SELECT p.prof_id,
       p.prof_fname,
       s.shift
FROM Professors p
JOIN Shift s
ON p.prof_id = s.prof_id;

QUERY 12
Department details with professor names
SELECT d.dept_name,
       p.prof_fname
FROM Departments d
JOIN Professors p
ON d.dept_id = p.dept_id;

QUERY 13
All professors and departments
SELECT p.prof_fname,
       d.dept_name
FROM Professors p
LEFT JOIN Departments d
ON p.dept_id = d.dept_id;

QUERY 14
SELECT p.prof_fname,
       d.dept_name,
       s.shift,
       p.salary
FROM Professors p
JOIN Departments d
ON p.dept_id=d.dept_id
JOIN Shift s
ON p.prof_id=s.prof_id
WHERE p.prof_id=101;

QUERY 15
Total professors in each department
SELECT dept_id,
       COUNT(*) AS Total_Professors
FROM Professors
GROUP BY dept_id;

QUERY 16
SELECT p.prof_id,
       d.dept_name
FROM Professors p
JOIN Departments d
ON p.dept_id=d.dept_id
WHERE d.dept_name='Computer';

QUERY 17
Departments where professors joined in 2015
SELECT DISTINCT d.dept_name
FROM Departments d
JOIN Professors p
ON d.dept_id=p.dept_id
WHERE EXTRACT(YEAR FROM p.doj)=2015;

QUERY 18
Create View
CREATE VIEW Professor_Shift_View AS

SELECT p.prof_id,
       p.prof_fname,
       s.shift,
       s.working_hours
FROM Professors p
JOIN Shift s
ON p.prof_id=s.prof_id;
