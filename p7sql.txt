CREATE TABLE Customer (
    cust_id INT PRIMARY KEY,
    cust_created_date DATE,
    cust_name VARCHAR(50),
    cust_phoneNo VARCHAR(15),
    cust_emailID VARCHAR(100),
    cust_city VARCHAR(50)
);

CREATE TABLE Product (
    product_id INT PRIMARY KEY,
    product_name VARCHAR(50),
    product_category VARCHAR(50),
    product_price DECIMAL(10,2)
);

CREATE TABLE Orders (
    order_id INT PRIMARY KEY,
    customer_id INT,
    product_id INT,
    order_date DATE,
    quantity INT,
    amount DECIMAL(10,2),

    FOREIGN KEY (customer_id)
    REFERENCES Customer(cust_id),

    FOREIGN KEY (product_id)
    REFERENCES Product(product_id)
);

Query 1: List Different Categories
SELECT DISTINCT product_category
FROM Product;

Query 2: Find Electronics Products Whose Price is Between 1000 and 5000
SELECT *
FROM Product
WHERE product_category = 'Electronics'
AND product_price BETWEEN 1000 AND 5000;

Query 3: Find Order Details Where Product Name Starts With E or C
SELECT o.*
FROM Orders o
JOIN Product p
ON o.product_id = p.product_id
WHERE p.product_name LIKE 'E%'
OR p.product_name LIKE 'C%';

Query 4: Calculate Total Amount Spent by Each Customer
SELECT c.cust_name,
       SUM(o.amount) AS total_spent
FROM Customer c
JOIN Orders o
ON c.cust_id = o.customer_id
GROUP BY c.cust_name;

Query 5: Display Products with Price Category
SELECT product_name,
       product_price,

       CASE
           WHEN product_price < 200
               THEN 'Cheap'

           WHEN product_price BETWEEN 200 AND 1000
               THEN 'Moderate'

           ELSE 'Expensive'
       END AS price_category

FROM Product;

Query 6: Delete Customers Who Have Never Placed Any Order
DELETE FROM Customer
WHERE cust_id NOT IN (
    SELECT customer_id
    FROM Orders
);
