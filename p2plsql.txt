Step 7: Create Procedure

CREATE OR REPLACE PROCEDURE proc_Grade
AS
    CURSOR c1 IS
        SELECT Cust_id, Name, Total_Purchase
        FROM Customer;

    v_class VARCHAR2(20);

BEGIN

    FOR rec IN c1 LOOP

        IF rec.Total_Purchase <= 20000 
           AND rec.Total_Purchase >= 10000 THEN
           
            v_class := 'Platinum';

        ELSIF rec.Total_Purchase BETWEEN 5000 AND 9999 THEN

            v_class := 'Gold';

        ELSIF rec.Total_Purchase BETWEEN 2000 AND 4999 THEN

            v_class := 'Silver';

        ELSE

            v_class := 'No Category';

        END IF;

        INSERT INTO Category
        VALUES(rec.Cust_id, rec.Name, v_class);

    END LOOP;

    COMMIT;

END;
