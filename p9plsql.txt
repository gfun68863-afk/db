CREATE OR REPLACE FUNCTION fn_creditscore(
    p_balance NUMBER,
    p_loan NUMBER
)
RETURN NUMBER

IS

    v_ratio NUMBER;
    v_score NUMBER;

BEGIN

    v_ratio := (p_loan / p_balance) * 100;

    IF v_ratio <= 20 THEN
        v_score := 5;

    ELSIF v_ratio <= 40 THEN
        v_score := 4;

    ELSIF v_ratio <= 50 THEN
        v_score := 3;

    ELSIF v_ratio <= 70 THEN
        v_score := 2;

    ELSIF v_ratio <= 90 THEN
        v_score := 1;

    ELSE
        v_score := 0;

    END IF;

    RETURN v_score;

END;
/

7. Create Stored Procedure Using Cursor
CREATE OR REPLACE PROCEDURE proc_creditscore

AS

    CURSOR c1 IS

        SELECT c.Cust_name,
               c.Balance,
               l.Loan_no,
               l.Amount
        FROM Customer c
        JOIN Loan l
        ON c.AccNo - 100 = l.Loan_no - 200;

    v_score NUMBER;

BEGIN

    FOR rec IN c1 LOOP

        v_score := fn_creditscore(
                        rec.Balance,
                        rec.Amount
                    );

        INSERT INTO Borrower
        VALUES(
            rec.Cust_name,
            rec.Loan_no,
            v_score
        );

    END LOOP;

    COMMIT;

END;
/
8. Execute Procedure
BEGIN
    proc_creditscore;
END;
