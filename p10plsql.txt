Step 7: Create Stored Procedure with Cursor
CREATE OR REPLACE PROCEDURE proc_customer_loan
AS
    CURSOR c1 IS
        SELECT c.Cust_name,
               c.Balance,
               l.Amount
        FROM Customer c
        JOIN Borrower b
        ON c.Cust_name = b.Cust_name
        JOIN Loan l
        ON b.Loan_no = l.Loan_no;
BEGIN
    FOR rec IN c1 LOOP
        DBMS_OUTPUT.PUT_LINE(
            'Customer Name : ' || rec.Cust_name
            || ' | Balance : ' || rec.Balance
            || ' | Loan Amount : ' || rec.Amount
        );
    END LOOP;
END;
/

Step 9: Execute Procedure
BEGIN
    proc_customer_loan;
END;
