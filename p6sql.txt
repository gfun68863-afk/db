CREATE TABLE Customer (
    cust_id INT PRIMARY KEY,
    cust_created_date DATE,
    cust_name VARCHAR(50),
    cust_phoneNo VARCHAR(15),
    cust_emailID VARCHAR(100),
    cust_city VARCHAR(50)
);

CREATE TABLE Product (
    product_id INT PRIMARY KEY,
    product_name VARCHAR(50),
    product_category VARCHAR(50),
    product_price DECIMAL(10,2)
);

CREATE TABLE Orders (
    order_id INT PRIMARY KEY,
    customer_id INT,
    product_id INT,
    order_date DATE,
    quantity INT,
    amount DECIMAL(10,2),

    FOREIGN KEY (customer_id)
    REFERENCES Customer(cust_id),

    FOREIGN KEY (product_id)
    REFERENCES Product(product_id)
);

Query 1: Find Products with Electronics, Stationary or Cloth Category
SELECT *
FROM Product
WHERE product_category IN ('Electronics','Stationary','Cloth');

Query 2: Find Customer Details Who Have Never Placed an Order
SELECT c.*
FROM Customer c
LEFT JOIN Orders o
ON c.cust_id = o.customer_id
WHERE o.order_id IS NULL;

Query 3: Find Names of 5 Most Expensive Products
SELECT product_name, product_price
FROM Product
ORDER BY product_price DESC
LIMIT 5;

Query 4: Display Customer Names Who Have Placed Orders Worth More Than 10000
SELECT DISTINCT c.cust_name
FROM Customer c
JOIN Orders o
ON c.cust_id = o.customer_id
WHERE o.amount > 10000;

Query 5: Increase Product Price of Electronics Category by 10%
UPDATE Product
SET product_price = product_price + (product_price * 0.10)
WHERE product_category = 'Electronics';

Query 6: Delete All Orders Placed Before 2020
DELETE FROM Orders
WHERE order_date < '2020-01-01';
