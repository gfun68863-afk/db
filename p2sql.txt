CREATE TABLE Student (
    stud_id INT PRIMARY KEY,
    stud_name VARCHAR(50),
    email VARCHAR(100),
    phone VARCHAR(15),
    dob DATE
);

CREATE TABLE Instructor (
    instructor_id INT PRIMARY KEY,
    instructor_name VARCHAR(50),
    salary DECIMAL(10,2)
);

CREATE TABLE Course (
    course_id INT PRIMARY KEY,
    course_name VARCHAR(50),
    credits INT,
    instructor_id INT,

    FOREIGN KEY (instructor_id)
    REFERENCES Instructor(instructor_id)
);

CREATE TABLE Enrollment (
    enrollment_id INT PRIMARY KEY,
    student_id INT,
    course_id INT,
    date_of_enrollment DATE,

    FOREIGN KEY (student_id)
    REFERENCES Student(stud_id),

    FOREIGN KEY (course_id)
    REFERENCES Course(course_id)
);

Query 1: List All Students Who Have Enrolled for Any Course
SELECT DISTINCT s.stud_name
FROM Student s
JOIN Enrollment e
ON s.stud_id = e.student_id;

Query 2: Find Students Enrolled in the Course "DBMS"
SELECT s.stud_name
FROM Student s
JOIN Enrollment e
ON s.stud_id = e.student_id
JOIN Course c
ON e.course_id = c.course_id
WHERE c.course_name = 'DBMS';

Query 3: Find Students Enrolled in More Than One Course
SELECT s.stud_name, COUNT(e.course_id) AS total_courses
FROM Student s
JOIN Enrollment e
ON s.stud_id = e.student_id
GROUP BY s.stud_name
HAVING COUNT(e.course_id) > 1;

Query 4: Find Students Enrolled in the Same Course as "John"
SELECT DISTINCT s2.stud_name
FROM Student s1
JOIN Enrollment e1
ON s1.stud_id = e1.student_id

JOIN Enrollment e2
ON e1.course_id = e2.course_id

JOIN Student s2
ON e2.student_id = s2.stud_id

WHERE s1.stud_name = 'John'
AND s2.stud_name <> 'John';

Query 5: Find Instructor(s) with Lowest Salary
SELECT instructor_name, salary
FROM Instructor
WHERE salary = (
    SELECT MIN(salary)
    FROM Instructor
);

Query 6: Create View of Instructors and Their Courses
CREATE VIEW Instructor_Course_View AS
SELECT i.instructor_name, c.course_name
FROM Instructor i
JOIN Course c
ON i.instructor_id = c.instructor_id;
