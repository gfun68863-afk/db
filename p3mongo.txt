Queries
i) Find information about all books of “Database”
db.books.find({
    Bookname: /Database/i
})

ii) Find book information of Author “Balkrishna” with Name “Fundamental C”
db.books.find({
    Bookname: "Fundamental C",
    Authors: "Balkrishna"
})

iii) Display total no. of books published in 2018
db.books.countDocuments({
    Year: 2018
})

iv) Find total count of books having rating greater than 4.5
db.books.countDocuments({
    book_rating: { $gt: 4.5 }
})

v) Delete books published before 1999
db.books.deleteMany({
    Year: { $lt: 1999 }
})

vi) Decrease no_of_copies by 5 for books published in year 2026
db.books.updateMany(
{
    Year: 2026
},
{
    $inc: {
        no_of_copies: -5
    }
}
)
