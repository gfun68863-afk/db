1. Single Field Index
db.students.createIndex({ roll_no: 1 })

2. Compound Index
db.students.createIndex({
    "address.city": 1,
    scholarship: 1
})

3. Ascending Index

db.students.createIndex({
    name: 1
})

db.students.find().sort({ name: 1 })

4. Descending Index
db.students.createIndex({
    admission_date: -1
})

db.students.find().sort({
    admission_date: -1
})

5. Multikey Index
{
   name: "Amit",
   hobbies: ["Cricket", "Music", "Coding"]
}

db.students.createIndex({
    hobbies: 1
})

db.students.find({
    hobbies: "Coding"
})

6. Text Index
db.students.createIndex({
    name: "text"
})

db.students.find({
    $text: {
        $search: "Amit"
    }
})

7. Unique Index
db.students.createIndex(
{
    roll_no: 1
},
{
    unique: true
})

8. Sparse Index
Example
db.students.createIndex(
{
    scholarship: 1
},
{
    sparse: true
})

9. TTL (Time To Live) Index
db.students.createIndex(
{
    exam_date: 1
},
{
    expireAfterSeconds: 3600
})
