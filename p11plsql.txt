Step 6: Create PL/SQL Function
CREATE OR REPLACE FUNCTION fn_creditscore(
    p_balance NUMBER,
    p_loan NUMBER
)
RETURN NUMBER
IS
    v_ratio NUMBER;
    v_score NUMBER;
BEGIN
    v_ratio := (p_loan / p_balance) * 100;

    IF v_ratio <= 20 THEN

        v_score := 5;

    ELSIF v_ratio <= 40 THEN

        v_score := 4;

    ELSIF v_ratio <= 50 THEN

        v_score := 3;

    ELSIF v_ratio <= 70 THEN

        v_score := 2;

    ELSIF v_ratio <= 90 THEN

        v_score := 1;

    ELSE

        v_score := 0;

    END IF;

    RETURN v_score;

END;
/
