1. Count Number of Courses Taught by Each Instructor
db.courses.aggregate([

{
    $group: {
        _id: "$instructor.name",
        total_courses: {
            $sum: 1
        }
    }
}

])

2. Calculate Average Marks Scored in Every Course
db.courses.aggregate([

{
    $unwind: "$enrollments"
},

{
    $group: {
        _id: "$title",
        average_marks: {
            $avg: "$enrollments.score"
        }
    }
}

])

3. List Faculty in Descending Order of Experience Within Each Department
db.courses.aggregate([

{
    $project: {
        department: "$instructor.department",
        faculty_name: "$instructor.name",
        experience: "$instructor.experience",
        designation: "$instructor.designation"
    }
},

{
    $sort: {
        department: 1,
        experience: -1
    }
}

])

4. List Students with Highest Scores
db.courses.aggregate([

{
    $unwind: "$enrollments"
},

{
    $sort: {
        "enrollments.score": -1
    }
},

{
    $limit: 1
},

{
    $project: {
        _id: 0,
        student_name: "$enrollments.student_name",
        score: "$enrollments.score",
        course: "$title"
    }
}

])

5. Calculate Pass Percentage in Each Course
db.courses.aggregate([

{
    $project: {
        title: 1,

        total_students: {
            $size: "$enrollments"
        },

        passed_students: {
            $size: {
                $filter: {
                    input: "$enrollments",
                    as: "student",
                    cond: {
                        $eq: ["$$student.pass_status", true]
                    }
                }
            }
        }
    }
},

{
    $project: {
        title: 1,

        pass_percentage: {
            $multiply: [
                {
                    $divide: [
                        "$passed_students",
                        "$total_students"
                    ]
                },
                100
            ]
        }
    }
}

])
