1. Single Field Index
db.movies.createIndex({
    movie_name: 1
})

2. Ascending Index
db.movies.createIndex({
    rating_star: 1
})

3. Descending Index
db.movies.createIndex({
    release_date: -1
})

4. Compound Index
Index on rating and collection amount
db.movies.createIndex({
    rating_star: 1,
    collection_amount: -1
})

db.movies.find({
    rating_star: { $gt: 4.0 }
}).sort({
    collection_amount: -1
})
5. Multikey Index
db.movies.createIndex({
    movie_cast: 1
})

db.movies.find({
    movie_cast: "Allu Arjun"
})

6. Text Index
Create Text Index on movie_name
db.movies.createIndex({
    movie_name: "text"
})
Query Example
db.movies.find({
    $text: {
        $search: "Avengers"
    }
})

7. Unique Index
db.movies.createIndex(
{
    movie_name: 1
},
{
    unique: true
})
